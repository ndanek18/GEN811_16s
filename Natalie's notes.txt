#activate environment

#cd into folder with fastq.gz
cd /home/users/nnd1011/GEN811_16s/data/fastq/20260306_16SV4_PE250/seq1_weekly_fresh

#make a manifest file
echo -e "sample-id\tforward-absolute-filepath\treverse-absolute-filepath" > manifest.tsv 

  
#make the manifest 
for file in *_R1.fastq.gz 

do 

    sample=$(basename "$file" _R1.fastq.gz) 

    forward=$(realpath "$file") 

    reverse=$(realpath "${sample}_R2.fastq.gz") 

echo -e "${sample}\t${forward}\t${reverse}" >> manifest.tsv 

done


#importing the manifest and the data in qime2 in a way that qime will recognize it

      qiime tools import \
  --type 'SampleData[PairedEndSequencesWithQuality]' \
  --input-path /home/users/nnd1011/GEN811_16s/data/fastq/20260306_16SV4_PE250/seq1_weekly_fresh/manifest.tsv \
  --output-path /home/users/nnd1011/GEN811_16s/data/fastq/20260306_16SV4_PE250/seq1_weekly_fresh/paired-end-demux.qza \
  --input-format PairedEndFastqManifestPhred33V2

#cutadapt to remove primers
qiime cutadapt trim-paired --i-demultiplexed-sequences paired-end-demux.qza --p-front-f GTGCCAGCMGCCGCGGTAA --p-front-r GGACTACHVGGGTWTCTAAT --o-trimmed-sequences trimmed-seqs.qza --verbose 


#summarize the data into a .qzv to be able to view when the quality scores drop off 
qiime demux summarize --i-data trimmed-seqs.qza --o-visualization trimmed-seqs.qzv 

#uploaded the .qzv into qiime and looked to see where the quality scores start to drop off

#cutting forward reads at 235 and reverse reads at 200
  qiime dada2 denoise-paired \
  --i-demultiplexed-seqs trimmed-seqs.qza \
  --p-trunc-len-f 235 \
  --p-trunc-len-r 200 \
  --p-n-threads 0 \
  --o-table feature-table.qza \
  --o-representative-sequences rep-seqs.qza \
  --o-denoising-stats denoising-stats.qza \
  --o-base-transition-stats base-transition-stats.qza \
  --verbose

  --o-feature-frequencies feature-frequencies.qza \
  --o-sample-frequencies sample-frequencies.qza \
  --o-summary feature-table.qzv
